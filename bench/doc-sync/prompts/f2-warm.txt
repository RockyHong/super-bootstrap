```sh
SLUG_HINT=""
slug_gap_hint() {
    SLUG_HINT=""
    slug_table "$1"
    local want="$2" line stripped oldifs="$IFS" oldflags="$-"
    # One fork rules the loop out: a table whose slugs are pure ASCII can hold no
    # dropped mark. That is every miss on an all-ASCII surface — the common case,
    # and the one where a per-slug fork would scale with the broken-link count.
    [ -n "$(printf '%s' "$SLUG_TABLE" | LC_ALL=C tr -d '\000-\177')" ] || return
    IFS='
'
    set -f
    for line in $SLUG_TABLE; do
        [ -n "$line" ] || continue
        stripped=$(printf '%s' "$line" | LC_ALL=C tr -d '\200-\377')
        [ "$stripped" = "$want" ] || continue
        SLUG_HINT=" (heading slugs as '$line' — SLUG_AWK's strip list is missing a mark GitHub drops)"
        break
    done
    case "$oldflags" in *f*) ;; *) set +f ;; esac
    IFS="$oldifs"
}
```

Each miss site now calls it and appends `%s` / `"$SLUG_HINT"` to its existing `printf`. `tests/doc-links.test.sh` gains a `BUG-055 follow-on` block of ten assertions across four fixtures.
